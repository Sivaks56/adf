CREATE TABLE ConfigTable
(
    ConfigID INT IDENTITY(1,1) PRIMARY KEY,

    SourceSchema VARCHAR(50) DEFAULT 'dbo',

    SourceTable VARCHAR(100),

    TargetSchema VARCHAR(50) DEFAULT 'dbo',

    TargetTable VARCHAR(100),

    LoadType VARCHAR(20),   -- FULL / INCREMENTAL

    IsActive BIT,

    WatermarkColumn VARCHAR(100),   -- Example: OrderDate etha vechi data last update aachinu find panna(date column name)

    WatermarkValue DATETIME NULL,   -- Last processed value where orderdate > last date iruntha atha mattum filter panni edukum ("10-07-2026")

    DeltaKeyColumn VARCHAR(100)    --- order_id tha primary key unique value vechi find pannalam
);
